# Daily Mix Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "Daily mix" of 5 variations at the top of the Openings panel (2 almost there,
2 review, 1 new), stable for the day with a "Change" button; "Continues into" capped at 3 rows.

**Architecture:** Pure `dailyMix()` + `seededRandom()` in `packages/core/src/lab/mix.ts`; a new
`DailyMix.svelte` rendered by `ExplorePanel` freezes the mix per `(day, shuffle)` and keeps the
shuffle counter in `localStorage`.

**Tech Stack:** TypeScript strict, Vitest, SvelteKit (Svelte 5 runes), Playwright.

**Spec:** `docs/superpowers/specs/2026-10-12-daily-mix-design.md`

## Global Constraints

- No new dependency; no DB/schema change; no generated file change.
- `packages/core` pure (no `Date.now()`, no storage): the day string comes from the web app.
- Every UI string via `t()`, keys added to `en.ts` and `it.ts` together:
  `mix.title` Daily mix / Mix del giorno; `mix.progress` {done}/{total} done today /
  {done}/{total} fatte oggi; `mix.shuffle` Change / Cambia; `mix.kind.almost` almost there /
  quasi fatta; `mix.kind.review` review / ripasso; `mix.kind.new` new / nuova;
  `lab.showAllContinuations` Show all ({n}) / Mostra tutte ({n}).
- `localStorage` key `pc.mix`, every access in try/catch.
- Known unrelated failures on this CRLF checkout: core `dataset.test.ts`, `graph.test.ts`,
  db `migrator.test.ts`.

## Review Focus

- Summaries load after the panel mounts: the mix must be recomputed once they arrive, not stay
  frozen on the empty-history fallback.
- Practising a mix line during the day must not replace it (stable goal); only ✓/dots update.
- `localStorage` throwing (private mode) → mix still renders with shuffle 0.
- Day rollover: a stored `{ day: yesterday, shuffle: 4 }` → today starts at shuffle 0.
- Long names in mix rows → ellipsis with full name in `title`.

```ts
import type { NameNode, NameTree } from './names';

export type MixKind = 'almost' | 'review' | 'new';

/** What the mix needs from a line's lab history (`lastAt` in Unix ms). */
export interface MixSummary {
  runs: number;
  cleanStreak: number;
  lastAt: number;
  lastClean: boolean;
}

export interface MixEntry {
  node: NameNode;
  kind: MixKind;
}

/** Deterministic PRNG (string hash + mulberry32): the same seed gives the same sequence. */
export function seededRandom(seed: string): () => number {
  let h = 1779033703 ^ seed.length;
  for (let i = 0; i < seed.length; i++) {
    h = Math.imul(h ^ seed.charCodeAt(i), 3432918353);
    h = (h << 13) | (h >>> 19);
  }
  let a = h >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** `n` items drawn without repetition from `pool` (not modified). */
function draw<T>(pool: readonly T[], n: number, random: () => number): T[] {
  const left = [...pool];
  const out: T[] = [];
  while (out.length < n && left.length)
    out.push(left.splice(Math.floor(random() * left.length), 1)[0]!);
  return out;
}

const POOL = 3;
const FALLBACK_POOL = 15;

/**
 * The day's practice mix: lines almost automatic, automatic lines to review, and new variations
 * next to practised ones. Stable for a given `day` and `shuffle`; each kind is drawn from its
 * best `POOL × slots` candidates. Without history, popular variations.
 */
export function dailyMix(opts: {
  tree: Pick<NameTree, 'all'>;
  summaryOf: (node: NameNode) => MixSummary | undefined;
  day: string;
  shuffle: number;
  masteredStreak: number;
  size?: number;
}): MixEntry[] {
  const size = opts.size ?? 5;
  const random = seededRandom(`${opts.day}:${opts.shuffle}`);
  const nodes = [...opts.tree.all()].filter((n) => n.kind !== 'family');
  const byPopularity = (a: NameNode, b: NameNode) =>
    b.lines - a.lines || a.key.localeCompare(b.key);

  const practised = nodes.flatMap((node) => {
    const s = opts.summaryOf(node);
    return s ? [{ node, s }] : [];
  });
  if (practised.length === 0) {
    const pool = [...nodes].sort(byPopularity).slice(0, FALLBACK_POOL);
    return draw(pool, size, random).map((node) => ({ node, kind: 'new' }));
  }

  const parents = new Set(practised.map((p) => p.node.parent));
  const pools: Record<MixKind, NameNode[]> = {
    almost: practised
      .filter((p) => p.s.cleanStreak < opts.masteredStreak)
      .sort((a, b) => b.s.lastAt - a.s.lastAt)
      .map((p) => p.node),
    review: practised
      .filter((p) => p.s.cleanStreak >= opts.masteredStreak)
      .sort((a, b) => a.s.lastAt - b.s.lastAt)
      .map((p) => p.node),
    new: nodes
      .filter((n) => n.parent !== null && parents.has(n.parent) && !opts.summaryOf(n))
      .sort(byPopularity),
  };
  const order: MixKind[] = ['almost', 'review', 'new'];
  const quota: Record<MixKind, number> = {
    almost: Math.round(size * 0.4),
    review: Math.round(size * 0.4),
    new: size - 2 * Math.round(size * 0.4),
  };

  const picked: Record<MixKind, NameNode[]> = { almost: [], review: [], new: [] };
  for (const kind of order) {
    picked[kind] = draw(pools[kind].slice(0, POOL * quota[kind]), quota[kind], random);
  }
  // unfilled slots go to the other kinds, in order
  let missing = size - order.reduce((n, k) => n + picked[k].length, 0);
  for (const kind of order) {
    if (missing <= 0) break;
    const rest = pools[kind].filter((n) => !picked[kind].includes(n));
    const extra = draw(rest.slice(0, POOL * missing), missing, random);
    picked[kind].push(...extra);
    missing -= extra.length;
  }
  return order.flatMap((kind) => picked[kind].map((node) => ({ node, kind })));
}
```

<script lang="ts">
  import {
    type MixEntry,
    type NameTree,
    type OpeningGraph,
    dailyMix,
    moverOfLast,
    openingLine,
  } from '@processchess/core';
  import type { LabRunSummary } from '@processchess/db';
  import Mastery from '$lib/components/Mastery.svelte';
  import SideMark from '$lib/components/SideMark.svelte';
  import { type LabItem, MASTERED_STREAK, lineKey } from '$lib/lab.svelte';
  import { t } from '$lib/i18n/index.svelte';
  import { now } from '$lib/time';
  import { untrack } from 'svelte';

  interface Props {
    graph: OpeningGraph;
    names: NameTree;
    summaries: Map<string, LabRunSummary>;
    onjump: (ucis: string[]) => void;
    onpractice: (queue: LabItem[], index: number) => void;
  }
  let { graph, names, summaries, onjump, onpractice }: Props = $props();

  const KEY = 'pc.mix';
  const localDay = (d: Date) =>
    `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
  const today = localDay(now());

  function readShuffle(): number {
    try {
      const v = JSON.parse(localStorage.getItem(KEY) ?? 'null') as {
        day?: string;
        shuffle?: number;
      } | null;
      return v?.day === today && typeof v.shuffle === 'number' ? v.shuffle : 0;
    } catch {
      return 0;
    }
  }
  let shuffle = $state(readShuffle());
  function change() {
    shuffle++;
    try {
      localStorage.setItem(KEY, JSON.stringify({ day: today, shuffle }));
    } catch {
      /* private mode: the counter lives in memory only */
    }
  }

  const lineOf = (node: { ucis: string[] }) => openingLine(graph, node.ucis);
  // frozen per (day, shuffle); recomputed once when the history first arrives
  const hasHistory = $derived(summaries.size > 0);
  const entries = $derived.by<MixEntry[]>(() => {
    void hasHistory;
    const s = shuffle;
    return untrack(() =>
      dailyMix({
        tree: names,
        day: today,
        shuffle: s,
        masteredStreak: MASTERED_STREAK,
        summaryOf: (n) => {
          const sum = summaries.get(lineKey(lineOf(n)));
          return sum && { ...sum, lastAt: sum.lastAt.getTime() };
        },
      }),
    );
  });
  const items = $derived<LabItem[]>(
    entries.map((e) => ({
      line: lineOf(e.node),
      eco: null,
      name: e.node.key,
      side: moverOfLast(e.node.ucis),
    })),
  );
  const doneToday = (it: LabItem) => {
    const s = summaries.get(lineKey(it.line));
    return !!s && s.lastClean && localDay(s.lastAt) === today;
  };
  const done = $derived(items.filter(doneToday).length);
</script>
